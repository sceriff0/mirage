# EXPORT_SPATIALDATA / JOIN_FLOWPATH: scverse serialization stack, CPU only.
#
# Exact pins, so a published .zarr can be traced to one image (the SpatialData on-disk
# format still changes between releases). To re-resolve:
#
#     docker build --platform linux/amd64 -f containers/spatialdata/Dockerfile -t mirage-spatialdata:lock .
#     docker run --rm mirage-spatialdata:lock python -m pip freeze | grep -iE \
#       '^(spatialdata|zarr|ome-zarr|anndata|geopandas|shapely|pyarrow|dask|tifffile|imagecodecs|scipy|pandas|numpy)=='
#
# Resolve on linux/amd64 (the shipped platform); an arm64 resolution can pick different
# wheels. Only the packages named here are pinned; the transitive closure is left to pip.
#
# No `-c constraints.txt`: this is the python 3.12 / zarr>=3 image, and the harmonised
# zarr==2.18.3 cannot run spatialdata>=0.8.0.
#
# No CI leg runs Python 3.12 with this stack; containers/spatialdata/smoke.sh is what
# exercises it, which is why it imports each of these packages by name.
spatialdata==0.9.0
zarr==3.4.2            # NOT the 2.18 the tiled image pins; spatialdata >= 0.8.0 needs 3.x
ome-zarr==0.21.0
anndata==0.13.4

# Shapes are GeoParquet-backed GeoDataFrames; GEOS comes from libgeos-c1v5 in the image.
geopandas==1.2.0
shapely==2.2.0
pyarrow==26.0.1         # GeoParquet writer for shapes/, and Parquet for any points/

# Lazy, chunked reads of the merged pyramid. Keep the `[array]` extra (pip freeze drops
# it): bin/export_spatialdata.py reads through dask.array.
dask[array]==2026.8.0
tifffile==2026.11.16
imagecodecs==2026.11.11     # zstd/LZW codecs used by MERGE_AND_PYRAMID's OME-TIFF

# cKDTree for the QC-residual -> cell_mask spatial join.
scipy==1.18.1
pandas==3.1.1
numpy==2.5.5
