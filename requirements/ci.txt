# CI's Python unit-test environment: the union of several production images' dependency
# sets, so it matches no single image. Every package CI shares with an image takes its
# version from that image's own file via `-r`; the two exceptions are noted where they sit.
#
# torch is not included: requirements/torch-cpu.txt carries an `--index-url` that would
# apply to the whole resolve, so CI installs it in a separate `pip install -r` step.
-c constraints.txt

# containers/segeval (CSE evaluation). Supplies numpy, scipy, pandas, scikit-image,
# scikit-learn, tifffile, xmltodict and matplotlib at the image's versions.
-r segeval.txt

# containers/stare. Supplies zarr, numcodecs and imagecodecs on the same harmonised pins.
# zarr and numcodecs must move together (newer numcodecs drops blosc.cbuffer_sizes that
# zarr 2.18 imports). Without imagecodecs, tifffile fails only on a write.
-r stare.txt

# ---- The STARE package: comes with `-r stare.txt` above, the version the image installs.

# ---- CI-only tooling -------------------------------------------------------------------
pytest
pytest-cov

# Named explicitly: tests/test_ci_job_hygiene.py importorskips it at module scope.
pyyaml

# ---- Runtime packages the suite needs that no `-r` above supplies ----------------------

# bin/convert_image.py's lazy read. Unpinned here: constraints.txt owns the version.
dask[array]

# The HDF5 test fixture and bin/convert_image.py's read_image_h5. Pinned here, not in
# constraints.txt, because only containers/convert installs it; matches that Dockerfile.
h5py==3.11.0

# Headless build (runners have no display libs). Unpinned: constraints.txt pins both
# opencv distribution names at the same release. opencv>=5 would need numpy>=2.
opencv-python-headless

# For tests/test_join_flowpath.py. anndata's production home is containers/spatialdata
# (python 3.12, zarr 3), which CI does not reproduce; zarr 3 breaks tifffile's aszarr, so
# CI caps anndata below 0.12 to work with zarr 2.18.3. Closing this gap needs a job that
# runs those tests inside the spatialdata image.
anndata<0.12
