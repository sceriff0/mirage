# Version authority for every package installed by more than one container image.
#
# Read by the Dockerfiles under containers/ and by the CI workflows, so each version is
# written once.
#   * Per-image files (stare.txt, segeval.txt, regqc.txt) open with `-c constraints.txt`;
#     pip resolves that path relative to the naming file, so Dockerfiles COPY both together.
#   * Dockerfiles that pin inline pass `-c /tmp/requirements/constraints.txt`. An inline pin
#     must agree with the constraint, or the build fails.
#   * requirements/ci.txt `-r`-includes the production files.
#
# tests/test_container_harmonisation.py reads this file as its harmonised table.
#
# Ceilings are the newest releases that support Python 3.10, except numpy, held at 1.x
# because the segmentation image's TensorFlow 2.15 base requires numpy<2.
numpy==1.26.4
tifffile==2025.5.10
imagecodecs==2025.3.30
zarr==2.18.3
numcodecs==0.13.1
pandas==2.3.3
scikit-image==0.25.2
scipy==1.15.3
dask==2026.7.1

# opencv under both distribution names at one release: regqc and merge install
# `opencv-python`, CI installs `opencv-python-headless`. The headless line constrains no
# image; it lets requirements/ci.txt name the package without a version.
opencv-python==4.10.0.82
opencv-python-headless==4.10.0.82
